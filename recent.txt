class CacheManager:
    def __init__(self, ttl_seconds=3600, max_size=1000):
        self.query_cache = TTLCache(maxsize=max_size, ttl=ttl_seconds)
        self.embedding_cache = LRUCache(maxsize=max_size)
        self.schema_cache = TTLCache(maxsize=max_size, ttl=ttl_seconds)
        self._lock = threading.Lock()

    def get_cached_query(self, query_key: str) -> Optional[str]:
        with self._lock:
            return self.query_cache.get(query_key)

    def cache_query(self, query_key: str, result: str):
        with self._lock:
            self.query_cache[query_key] = result

    def get_cached_embedding(self, text_key: str) -> Optional[List[float]]:
        with self._lock:
            return self.embedding_cache.get(text_key)

    def cache_embedding(self, text_key: str, embedding: List[float]):
        with self._lock:
            self.embedding_cache[text_key] = embedding

    def get_cached_schema(self, schema_key: str) -> Optional[List[Dict]]:
        with self._lock:
            return self.schema_cache.get(schema_key)

    def cache_schema(self, schema_key: str, schemas: List[Dict]):
        with self._lock:
            self.schema_cache[schema_key] = schemas

class VectorStoreManager:
    def __init__(self, chroma_client, collection_prefix: str):
        self.client = chroma_client
        self.prefix = collection_prefix
        self.collections = {}
        self.initialize_collections()

    def initialize_collections(self):
        collection_types = ["schema", "query", "metadata"]
        for ctype in collection_types:
            name = f"{self.prefix}_{ctype}"
            try:
                self.collections[ctype] = self.client.get_or_create_collection(
                    name=name,
                    metadata={"type": ctype, "created_at": datetime.now().isoformat()}
                )
                logger.info(f"✅ Successfully initialized collection: {name}")
            except Exception as e:
                logger.error(f"❌ Error initializing collection {name}: {str(e)}")
                raise

    async def add_vectors(self, collection_type: str, ids: List[str], 
                         embeddings: List[List[float]], documents: List[str], 
                         metadatas: List[Dict] = None):
        if collection_type not in self.collections:
            raise ValueError(f"Invalid collection type: {collection_type}")
        
        collection = self.collections[collection_type]
        batch_size = 100
        
        for i in range(0, len(ids), batch_size):
            batch_ids = ids[i:i + batch_size]
            batch_embeddings = embeddings[i:i + batch_size]
            batch_documents = documents[i:i + batch_size]
            batch_metadatas = metadatas[i:i + batch_size] if metadatas else None
            
            collection.add(
                ids=batch_ids,
                embeddings=batch_embeddings,
                documents=batch_documents,
                metadatas=batch_metadatas
            )

    async def upsert_vectors(self, collection_type: str, ids: List[str], 
                            embeddings: List[List[float]], documents: List[str], 
                            metadatas: List[Dict] = None):
        if collection_type not in self.collections:
            raise ValueError(f"Invalid collection type: {collection_type}")
        
        collection = self.collections[collection_type]
        batch_size = 100
        
        for i in range(0, len(ids), batch_size):
            batch_ids = ids[i:i + batch_size]
            batch_embeddings = embeddings[i:i + batch_size]
            batch_documents = documents[i:i + batch_size]
            batch_metadatas = metadatas[i:i + batch_size] if metadatas else None
            
            collection.upsert(
                ids=batch_ids,
                embeddings=batch_embeddings,
                documents=batch_documents,
                metadatas=batch_metadatas
            )

class EnhancedOllamaChat(Ollama):
    def __init__(self, config: Dict = None):
        config = config or {'model': 'qwen2.5-coder:14b'}
        self.config = config
        self.dialect = 'mssql'
        self.max_tokens = 1024
        self.static_documentation = {}
        super().__init__(config=config)
        self.training_data = {}
        self.encoder = SentenceTransformer('paraphrase-MiniLM-L6-v2')

    def generate_text(self, prompt: str) -> str:
        try:
            response = self.submit_prompt(prompt)
            if isinstance(response, str):
                return response.strip()
            return str(response)
        except Exception as e:
            logger.error(f"Error generating text: {str(e)}")
            raise

    def generate_sql(self, prompt: str) -> str:
        try:
            response = super().generate_sql(prompt)
            if isinstance(response, str):
                return response.strip()
            elif isinstance(response, list) and len(response) > 0:
                first_element = response[0]
                if isinstance(first_element, str):
                    return first_element.strip()
                elif isinstance(first_element, dict) and 'text' in first_element:
                    return first_element['text'].strip()
                else:
                    raise ValueError(f"Unexpected response format: {response}")
            elif isinstance(response, dict) and 'text' in response:
                return response['text'].strip()
            else:
                raise ValueError(f"Unexpected response format: {response}")
        except Exception as e:
            logger.error(f"Error generating SQL: {str(e)}")
            raise

    def add_ddl(self, ddl: str) -> str:
        ddl_id = hashlib.md5(ddl.encode()).hexdigest()
        self.training_data[ddl_id] = {"type": "ddl", "content": ddl}
        return ddl_id

    def add_documentation(self, documentation: str) -> str:
        doc_id = hashlib.md5(documentation.encode()).hexdigest()
        self.training_data[doc_id] = {"type": "documentation", "content": documentation}
        return doc_id

    def add_question_sql(self, question: str, sql: str) -> str:
        pair_id = hashlib.md5(f"{question}:{sql}".encode()).hexdigest()
        self.training_data[pair_id] = {
            "type": "question_sql",
            "question": question,
            "sql": sql
        }
        return pair_id

    def generate_embedding(self, text: str) -> List[float]:
        return self.encoder.encode(text).tolist()

    def get_related_ddl(self, text: str, n: int = 5) -> List[str]:
        ddls = [item["content"] for item in self.training_data.values() 
                if item["type"] == "ddl"]
        if not ddls:
            return []
        
        text_embedding = self.generate_embedding(text)
        ddl_embeddings = [self.generate_embedding(ddl) for ddl in ddls]
        
        similarities = [
            cosine_similarity(text_embedding, ddl_embedding) 
            for ddl_embedding in ddl_embeddings
        ]
        
        sorted_pairs = sorted(zip(ddls, similarities), key=lambda x: x[1], reverse=True)
        return [ddl for ddl, _ in sorted_pairs[:n]]

    def get_related_documentation(self, text: str, n: int = 5) -> List[str]:
        docs = [item["content"] for item in self.training_data.values() 
               if item["type"] == "documentation"]
        if not docs:
            return []
        
        text_embedding = self.generate_embedding(text)
        doc_embeddings = [self.generate_embedding(doc) for doc in docs]
        
        similarities = [
            cosine_similarity(text_embedding, doc_embedding) 
            for doc_embedding in doc_embeddings
        ]
        
        sorted_pairs = sorted(zip(docs, similarities), key=lambda x: x[1], reverse=True)
        return [doc for doc, _ in sorted_pairs[:n]]

    def get_similar_question_sql(self, text: str, n: int = 5) -> List[Tuple[str, str]]:
        pairs = [(item["question"], item["sql"]) 
                for item in self.training_data.values() 
                if item["type"] == "question_sql"]
        if not pairs:
            return []
        
        text_embedding = self.generate_embedding(text)
        pair_embeddings = [
            self.generate_embedding(q) for q, _ in pairs
        ]
        
        similarities = [
            cosine_similarity(text_embedding, pair_embedding) 
            for pair_embedding in pair_embeddings
        ]
        
        sorted_pairs = sorted(zip(pairs, similarities), key=lambda x: x[1], reverse=True)
        return [pair for pair, _ in sorted_pairs[:n]]

    def get_training_data(self) -> Dict:
        return self.training_data

    def remove_training_data(self, data_id: str) -> bool:
        if data_id in self.training_data:
            del self.training_data[data_id]
            return True
        return False

def cosine_similarity(v1: List[float], v2: List[float]) -> float:
    dot_product = sum(a * b for a, b in zip(v1, v2))
    norm_v1 = math.sqrt(sum(a * a for a in v1))
    norm_v2 = math.sqrt(sum(b * b for b in v2))
    return dot_product / (norm_v1 * norm_v2) if norm_v1 * norm_v2 != 0 else 0

    def _compute_schema_hash(self, schema: Dict) -> str:
        """Compute a hash of the schema for change detection."""
        schema_json = json.dumps(schema, sort_keys=True)
        return hashlib.md5(schema_json.encode()).hexdigest()

    @retry(stop=stop_after_attempt(3), wait=wait_exponential(multiplier=1, min=4, max=10))
    def store_database_schema(self):
        """Store database schema in ChromaDB with incremental updates."""
        try:
            # Get current schema from database
            connection = self._create_db_connection()
            cursor = connection.cursor()

            # Fetch all tables
            cursor.execute("""
                SELECT TABLE_NAME 
                FROM INFORMATION_SCHEMA.TABLES 
                WHERE TABLE_TYPE = 'BASE TABLE'
            """)
            current_tables = {row[0] for row in cursor.fetchall()}

            # Get existing schemas from ChromaDB
            existing_docs = self.vector_store.collections["schema"].get(include=["documents", "metadatas"])
            existing_schemas = {}
            for doc_id, doc in zip(existing_docs["ids"], existing_docs["documents"]):
                try:
                    schema = json.loads(doc)
                    existing_schemas[doc_id] = schema
                except json.JSONDecodeError:
                    continue

            # Identify tables to add, update, or remove
            tables_to_add = current_tables - set(existing_schemas.keys())
            tables_to_update = set()
            tables_to_remove = set(existing_schemas.keys()) - current_tables

            # Check for schema changes in existing tables
            for table_name in current_tables:
                if table_name in existing_schemas:
                    # Fetch current table schema
                    cursor.execute("""
                        SELECT 
                            COLUMN_NAME, DATA_TYPE, IS_NULLABLE, COLUMN_DEFAULT, 
                            CHARACTER_MAXIMUM_LENGTH
                        FROM INFORMATION_SCHEMA.COLUMNS
                        WHERE TABLE_NAME = ?
                    """, table_name)
                    current_columns = {row[0]: {
                        "data_type": row[1],
                        "is_nullable": row[2],
                        "default_value": row[3],
                        "character_max_length": row[4]
                    } for row in cursor.fetchall()}

                    # Compare with existing schema
                    existing_columns = {
                        col["column_name"]: {
                            "data_type": col["data_type"],
                            "is_nullable": col["is_nullable"],
                            "default_value": col["default_value"],
                            "character_max_length": col["character_max_length"]
                        } for col in existing_schemas[table_name]["columns"]
                    }

                    if current_columns != existing_columns:
                        tables_to_update.add(table_name)

            # Process schema changes
            updates = []
            deletes = []

            # Handle new tables
            for table_name in tables_to_add:
                schema = self._fetch_table_schema(cursor, table_name)
                if schema:
                    schema_json = json.dumps(schema)
                    schema_embedding = self.get_embedding(schema_json)
                    updates.append({
                        "id": table_name,
                        "embedding": schema_embedding,
                        "document": schema_json,
                        "metadata": {"db_name": self.db_name}
                    })

            # Handle updated tables
            for table_name in tables_to_update:
                schema = self._fetch_table_schema(cursor, table_name)
                if schema:
                    schema_json = json.dumps(schema)
                    schema_embedding = self.get_embedding(schema_json)
                    updates.append({
                        "id": table_name,
                        "embedding": schema_embedding,
                        "document": schema_json,
                        "metadata": {"db_name": self.db_name}
                    })

            # Handle removed tables
            deletes = list(tables_to_remove)

            # Batch process updates and deletes
            if updates:
                batch_ids = [item["id"] for item in updates]
                batch_embeddings = [item["embedding"] for item in updates]
                batch_documents = [item["document"] for item in updates]
                batch_metadatas = [item["metadata"] for item in updates]

                asyncio.run(self.vector_store.upsert_vectors(
                    collection_type="schema",
                    ids=batch_ids,
                    embeddings=batch_embeddings,
                    documents=batch_documents,
                    metadatas=batch_metadatas
                ))
                logger.info(f"Updated {len(batch_ids)} tables in ChromaDB")

            if deletes:
                self.vector_store.collections["schema"].delete(ids=deletes)
                logger.info(f"Removed {len(deletes)} tables from ChromaDB")

            cursor.close()
            connection.close()
            logger.info(f"✅ Successfully updated schema for database '{self.db_name}'")
            
        except Exception as e:
            logger.error(f"❌ Error while updating database schema for {self.db_name}: {e}")
            raise

    def _fetch_table_schema(self, cursor, table_name: str) -> Dict:
        """Fetch complete schema information for a single table."""
        try:
            # Fetch columns
            cursor.execute("""
                SELECT 
                    COLUMN_NAME, DATA_TYPE, IS_NULLABLE, COLUMN_DEFAULT, 
                    CHARACTER_MAXIMUM_LENGTH
                FROM INFORMATION_SCHEMA.COLUMNS
                WHERE TABLE_NAME = ?
            """, table_name)
            columns = []
            for row in cursor.fetchall():
                columns.append({
                    "column_name": row[0],
                    "data_type": row[1],
                    "is_nullable": row[2] == "YES",
                    "default_value": row[3],
                    "character_max_length": row[4],
                    "is_primary_key": False,
                    "is_foreign_key": False,
                    "foreign_key_reference": None
                })

            # Fetch primary keys
            cursor.execute("""
                SELECT COLUMN_NAME
                FROM INFORMATION_SCHEMA.KEY_COLUMN_USAGE
                WHERE TABLE_NAME = ? AND CONSTRAINT_NAME LIKE '%PK%'
            """, table_name)
            primary_keys = [row[0] for row in cursor.fetchall()]
            for col in columns:
                if col["column_name"] in primary_keys:
                    col["is_primary_key"] = True

            # Fetch foreign keys
            cursor.execute("""
                SELECT 
                    COL_NAME(fc.parent_object_id, fc.parent_column_id) AS COLUMN_NAME,
                    OBJECT_NAME(fk.referenced_object_id) AS REFERENCED_TABLE_NAME,
                    COL_NAME(fc.referenced_object_id, fc.referenced_column_id) AS REFERENCED_COLUMN_NAME
                FROM sys.foreign_keys AS fk
                INNER JOIN sys.foreign_key_columns AS fc ON fk.object_id = fc.constraint_object_id
                WHERE OBJECT_NAME(fk.parent_object_id) = ?
            """, table_name)
            foreign_keys = {}
            for row in cursor.fetchall():
                ref = f"{row[1]}({row[2]})"
                foreign_keys[row[0]] = ref
                for col in columns:
                    if col["column_name"] == row[0]:
                        col["is_foreign_key"] = True
                        col["foreign_key_reference"] = ref

            return {
                "table_name": table_name,
                "columns": columns,
                "primary_keys": primary_keys,
                "foreign_keys": foreign_keys,
                "table_description": f"Table {table_name} in database {self.db_name}",
                "relationships": foreign_keys
            }
        except Exception as e:
            logger.error(f"Error fetching schema for table {table_name}: {e}")
            return None

    async def process_query(self, query: str, context: QueryContext) -> str:
        try:
            cache_key = hashlib.md5(f"{query}:{self.db_name}".encode()).hexdigest()
            cached_result = self.cache_manager.get_cached_query(cache_key)
            if cached_result:
                logger.info(f"Cache hit for query: {query[:50]}...")
                return cached_result
            
            await self.query_queue.put((context.priority, (query, context)))
            sql_query = await self._process_query_with_priority(query, context)
            self.cache_manager.cache_query(cache_key, sql_query)
            return sql_query
            
        except Exception as e:
            logger.error(f"Error processing query: {str(e)}")
            raise

    async def _process_query_with_priority(self, query: str, context: QueryContext) -> str:
        try:
            intent = await self._analyze_query_intent(query)
            schemas = await self._hybrid_schema_search(query, intent)
            if not schemas:
                return "Error: No relevant schema found for your query."
            
            if context.query_type == QueryType.SIMPLE:
                sql = await self._generate_simple_query(query, schemas, intent)
            elif context.query_type == QueryType.COMPLEX:
                sql = await self._generate_complex_query(query, schemas, intent)
            elif context.query_type == QueryType.ANALYTICAL:
                sql = await self._generate_analytical_query(query, schemas, intent)
            else:
                sql = await self._generate_default_query(query, schemas, intent)
            
            return sql
            
        except Exception as e:
            logger.error(f"Error in query processing: {str(e)}")
            raise

    async def _hybrid_schema_search(self, query: str, intent: Dict) -> List[Dict]:
        try:
            cache_key = f"{self.db_name}:{query}:hybrid_search"
            cached_schemas = self.cache_manager.get_cached_schema(cache_key)
            if cached_schemas:
                logger.info("Returning schemas from cache")
                return cached_schemas

            query_embedding = await self._get_embedding_async(query)
            sparse_vec = self.sparse_encoder.fit_transform([query]).toarray()[0]
            dense_results = await self._dense_search(query_embedding)
            sparse_results = await self._sparse_search(sparse_vec)
            combined_results = self._combine_search_results(
                dense_results, sparse_results, weights=[0.7, 0.3]
            )
            final_schemas = await self._filter_and_rank_schemas(combined_results, intent)

            self.cache_manager.cache_schema(cache_key, final_schemas)
            return final_schemas[:3]  # Limit to top 3 relevant schemas
            
        except Exception as e:
            logger.error(f"Error in hybrid search: {str(e)}")
            raise

    async def _dense_search(self, query_embedding: List[float], top_k: int = 5) -> List[Dict]:
        try:
            results = self.vector_store.collections["schema"].query(
                query_embeddings=[query_embedding],
                n_results=top_k
            )
            return [json.loads(doc) for doc in results["documents"][0]]
        except Exception as e:
            logger.error(f"Error in dense search: {str(e)}")
            raise

    async def _sparse_search(self, sparse_vec: np.ndarray, top_k: int = 5) -> List[Dict]:
        try:
            all_docs = self.vector_store.collections["schema"].get()
            similarities = []
            for doc in all_docs["documents"]:
                doc_vec = self.sparse_encoder.transform([doc]).toarray()[0]
                similarity = np.dot(sparse_vec, doc_vec)
                similarities.append((similarity, json.loads(doc)))
            similarities.sort(reverse=True)
            return [doc for _, doc in similarities[:top_k]]
            
        except Exception as e:
            logger.error(f"Error in sparse search: {str(e)}")
            raise

    def _combine_search_results(self, dense_results: List[Dict], 
                              sparse_results: List[Dict], 
                              weights: List[float]) -> List[Dict]:
        combined_scores = {}
        for i, result in enumerate(dense_results):
            score = weights[0] * (1.0 - (i / len(dense_results)))
            combined_scores[result["table_name"]] = score
        
        for i, result in enumerate(sparse_results):
            score = weights[1] * (1.0 - (i / len(sparse_results)))
            if result["table_name"] in combined_scores:
                combined_scores[result["table_name"]] += score
            else:
                combined_scores[result["table_name"]] = score
        
        sorted_results = sorted(
            combined_scores.items(), 
            key=lambda x: x[1], 
            reverse=True
        )
        
        final_results = []
        for table_name, _ in sorted_results:

                ids=[table_name]
            )
            if schema["documents"]:
                final_results.append(json.loads(schema["documents"][0]))
        
        return final_results

    async def _analyze_query_intent(self, query: str) -> Dict:
        """Analyze query intent and extract relevant information."""
        try:
            # Check cache first
            cache_key = hashlib.md5(query.lower().strip().encode()).hexdigest()
            cached_intent = self.cache_manager.get_cached_query(cache_key)
            if cached_intent:
                return json.loads(cached_intent)

            intent = {
                "action_type": "select",  # select, insert, update, delete
                "query_complexity": "simple",  # simple, moderate, complex
                "temporal_aspect": None,  # current, historical, future
                "aggregation_level": None,  # none, basic, advanced
                "join_complexity": "none",  # none, simple, complex
                "key_terms": set(),
                "entities": set(),
                "conditions": [],
                "aggregations": [],
                "time_range": None,
                "order": None,
                "limit": None,
                "tables": [],
                "joins": [],
                "columns": set(),
                "estimated_cost": 0,
                "performance_hints": [],
                "partition_aware": False
            }

            # Standardize query processing
            query_lower = query.lower().strip()
            
            # Analyze action type
            action_keywords = {
                "select": ["show", "list", "get", "find", "display", "retrieve"],
                "insert": ["add", "create", "insert", "new"],
                "update": ["update", "modify", "change", "edit"],
                "delete": ["delete", "remove", "drop"]
            }
            for action, keywords in action_keywords.items():
                if any(keyword in query_lower for keyword in keywords):
                    intent["action_type"] = action
                    break

            # Analyze query complexity
            complexity_indicators = {
                "complex": ["group by", "having", "partition by", "window", "rank", "dense_rank"],
                "moderate": ["join", "where", "order by", "distinct"],
                "simple": ["select", "from"]
            }
            for complexity, indicators in complexity_indicators.items():
                if any(indicator in query_lower for indicator in indicators):
                    intent["query_complexity"] = complexity
                    break

            # Analyze temporal aspect
            temporal_keywords = {
                "current": ["current", "now", "today", "present"],
                "historical": ["history", "past", "previous", "last"],
                "future": ["future", "next", "upcoming", "forecast"]
            }
            for aspect, keywords in temporal_keywords.items():
                if any(keyword in query_lower for keyword in keywords):
                    intent["temporal_aspect"] = aspect
                    break

            # Analyze aggregation level
            aggregation_keywords = {
                "advanced": ["percentile", "median", "mode", "stddev", "variance"],
                "basic": ["sum", "avg", "count", "min", "max"]
            }
            for level, keywords in aggregation_keywords.items():
                if any(keyword in query_lower for keyword in keywords):
                    intent["aggregation_level"] = level
                    break

            # Analyze join complexity
            join_patterns = {
                "complex": ["left join", "right join", "full join", "cross join"],
                "simple": ["join", "inner join"]
            }
            for complexity, patterns in join_patterns.items():
                if any(pattern in query_lower for pattern in patterns):
                    intent["join_complexity"] = complexity
                    break

            # Get schema information
            schema_docs = self.vector_store.collections["schema"].get()
            table_names = set()
            table_schemas = {}
            
            # Process schemas consistently
            for doc in schema_docs["documents"]:
                try:
                    schema = json.loads(doc)
                    if "table_name" in schema:
                        table_name = schema["table_name"]
                        table_names.add(table_name.lower())
                        table_schemas[table_name] = schema
                except json.JSONDecodeError:
                    continue

            # Standardized table detection
            detected_tables = set()
            for table_name, schema in table_schemas.items():
                table_name_lower = table_name.lower()
                if re.search(rf'\b{re.escape(table_name_lower)}\b', query_lower):
                    detected_tables.add(table_name)
                else:
                    synonyms = self._generate_table_synonyms(table_name)
                    for synonym in synonyms:
                        if synonym and re.search(rf'\b{re.escape(synonym.lower())}\b', query_lower):
                            detected_tables.add(table_name)
                            break

            # Process tables in consistent order
            intent["tables"] = sorted(list(detected_tables))

            # Standardized relationship detection
            for table_name in intent["tables"]:
                if table_name in table_schemas:
                    schema = table_schemas[table_name]
                    if "relationships" in schema:
                        for fk_col, ref in schema["relationships"].items():
                            ref_table = ref.split('(')[0]
                            ref_col = ref.split('(')[1].rstrip(')')
                            if ref_table in table_names:
                                join_info = {
                                    "source_table": table_name,
                                    "source_column": fk_col,
                                    "target_table": ref_table,
                                    "target_column": ref_col,
                                    "estimated_cost": self._estimate_join_cost(table_name, ref_table)
                                }
                                if join_info not in intent["joins"]:
                                    intent["joins"].append(join_info)

            # Standardized column extraction
            column_pattern = r'\b([A-Za-z_][A-Za-z0-9_]*)\b'
            words = set(re.findall(column_pattern, query_lower))
            
            for table_name, schema in table_schemas.items():
                if "columns" in schema:
                    for col in schema["columns"]:
                        col_name = col["column_name"].lower()
                        if col_name in words or any(synonym.lower() in words for synonym in self._generate_column_synonyms(col_name)):
                            intent["columns"].add(col["column_name"])
                            if col.get("is_foreign_key"):
                                ref = col.get("foreign_key_reference")
                                if ref:
                                    ref_table = ref.split('(')[0]
                                    if ref_table not in intent["tables"]:
                                        intent["tables"].append(ref_table)

            # Standardize column filtering
            sql_keywords = {"select", "from", "where", "join", "group", "order", "having", "limit", "details", "all"}
            intent["columns"] = sorted(list({col for col in intent["columns"] if col.lower() not in table_names and col.lower() not in sql_keywords}))

            # Standardize key terms
            common_words = {
                "the", "a", "an", "and", "or", "but", "in", "on", "at", "to", "for", 
                "with", "by", "from", "up", "about", "into", "over", "after", "show", 
                "give", "list", "display", "me", "get", "fetch", "find", "search"
            }
            intent["key_terms"] = sorted(list({word.lower() for word in words if word.lower() not in common_words and word.lower() not in sql_keywords}))

            # Standardize conditions
            condition_markers = ["where", "with", "having", "contains", "equals", "equal to", 
                               "greater than", "less than", "between", "is", "not", "="]
            for marker in condition_markers:
                if marker in query_lower:
                    condition_idx = query_lower.find(marker)
                    if condition_idx != -1:
                        condition_part = query[condition_idx:].strip()
                        if condition_part not in intent["conditions"]:
                            intent["conditions"].append(condition_part)

            # Calculate cost consistently
            intent["estimated_cost"] = self._calculate_query_cost(intent)

            # Cache the standardized intent
            self.cache_manager.cache_query(cache_key, json.dumps(intent, default=str))

            return intent
        except Exception as e:
            logger.error(f"❌ Error in _analyze_query_intent: {str(e)}")
            raise

    def _get_table_sizes(self) -> Dict[str, float]:
        """Get approximate sizes of tables in MB."""
        try:
            connection = self._create_db_connection()
            cursor = connection.cursor()
            
            cursor.execute("""
                SELECT 
                    t.NAME AS TableName,
                    s.Name AS SchemaName,
                    p.rows AS RowCounts,
                    SUM(a.total_pages) * 8 AS TotalSpaceKB
                FROM 
                    sys.tables t
                INNER JOIN      
                    sys.indexes i ON t.OBJECT_ID = i.object_id
                INNER JOIN 
                    sys.partitions p ON i.object_id = p.OBJECT_ID AND i.index_id = p.index_id
                INNER JOIN 
                    sys.allocation_units a ON p.partition_id = a.container_id
                LEFT OUTER JOIN 
                    sys.schemas s ON t.schema_id = s.schema_id
                WHERE 
                    t.NAME NOT LIKE 'dt%' 
                    AND t.is_ms_shipped = 0
                    AND i.OBJECT_ID > 255 
                GROUP BY 
                    t.Name, s.Name, p.Rows
                ORDER BY 
                    t.Name
            """)
            
            table_sizes = {}
            for row in cursor.fetchall():
                table_name = row[0]
                size_mb = row[3] / 1024.0  # Convert KB to MB
                table_sizes[table_name] = size_mb
                
            cursor.close()
            connection.close()
            return table_sizes
            
        except Exception as e:
            logger.error(f"Error getting table sizes: {e}")
            return {}

    def _estimate_join_cost(self, table1: str, table2: str) -> float:
        """Estimate the cost of joining two tables."""
        try:
            table_sizes = self._get_table_sizes()
            size1 = table_sizes.get(table1, 0)
            size2 = table_sizes.get(table2, 0)
            
            # Simple cost estimation based on table sizes
            if size1 == 0 or size2 == 0:
                return 1.0
                
            # Cost increases with the product of table sizes
            return min(1.0, (size1 * size2) / 1000000.0)
        except Exception as e:
            logger.error(f"Error estimating join cost: {e}")
            return 1.0

    def _calculate_query_cost(self, intent: Dict) -> float:
        """Calculate estimated cost of the query based on intent."""
        try:
            cost = 0.0
            table_sizes = self._get_table_sizes()
            
            # Base cost for each table
            for table in intent["tables"]:
                size_mb = table_sizes.get(table, 0)
                cost += size_mb / 100.0  # Normalize to reasonable range
            
            # Additional cost for joins
            for join in intent["joins"]:
                cost += join.get("estimated_cost", 1.0)
            
            # Cost for aggregations
            if intent["aggregations"]:
                cost *= 1.5
            
            # Cost for ordering
            if intent["order"]:
                cost *= 1.2
            
            # Cost for time-based analysis
            if intent["time_range"]:
                cost *= 1.3
            
            return min(100.0, cost)  # Cap at 100
        except Exception as e:
            logger.error(f"Error calculating query cost: {e}")
            return 1.0

            self._format_schema_context(schemas),

    async def _generate_sql_with_validation(self, prompt: str) -> str:
        try:
            query_intent = await self._analyze_query_intent(prompt)
            if query_intent["tables"] and len(query_intent["tables"]) > 1:
                tables = query_intent["tables"]
                joins = query_intent["joins"]
                sql_parts = [f"SELECT {tables[0]}.*"]
                for table in tables[1:]:
                    sql_parts[0] += f", {table}.*"
                sql_parts.append(f"FROM {tables[0]}")
                for join in joins:
                    sql_parts.append(
                        f"JOIN {join['target_table']} ON "
                        f"{join['source_table']}.{join['source_column']} = "
                        f"{join['target_table']}.{join['target_column']}"
                    )
                sql = "\n".join(sql_parts)
            else:
                sql = await self.generate_sql(prompt, allow_llm_to_see_data=True).strip()
            
            if "JOIN" in sql.upper() and len(sql) > 500:
                sql = self._add_optimization_hints(sql)
            
            return sql.strip()
            
        except Exception as e:
            logger.error(f"Error generating SQL with validation: {str(e)}")
            raise

    def _add_optimization_hints(self, sql: str) -> str:
        if "ORDER BY" in sql.upper():
            sql += "\nOPTION (RECOMPILE)"
        return sql

    def _generate_table_synonyms(self, table_name: str) -> List[str]:
        synonyms = []
        synonyms.append(table_name)
        clean_name = re.sub(r'[^a-zA-Z0-9]', ' ', table_name).strip()
        if clean_name != table_name:
            synonyms.append(clean_name)
        camel_split = re.sub(r'([a-z])([A-Z])', r'\1 \2', table_name)
        if camel_split != table_name:
            synonyms.append(camel_split)
        if table_name.startswith("tbl"):
            synonyms.append(table_name[3:])
        singular = p.singular_noun(table_name)
        if singular:
            synonyms.append(singular)
        else:
            plural = p.plural(table_name)
            synonyms.append(plural)
        return list(set(synonyms))

    def _generate_column_synonyms(self, column_name: str) -> List[str]:
        synonyms = []
        synonyms.append(column_name)
        if column_name.startswith("col"):
            synonyms.append(column_name[3:])
        if column_name.endswith("ID") or column_name.endswith("Id"):
            synonyms.append(column_name[:-2])
        clean_name = re.sub(r'_', ' ', column_name).strip()
        if clean_name != column_name:
            synonyms.append(clean_name)
        camel_split = re.sub(r'([a-z])([A-Z])', r'\1 \2', column_name)
        if camel_split != column_name:
            synonyms.append(camel_split)
        return list(set(synonyms))

    def _normalize_table_name(self, table_name: str) -> str:
        """Normalize table name by removing spaces, underscores, and standardizing case."""
        # Remove spaces and underscores
        normalized = re.sub(r'[\s_]+', '', table_name.lower())
        
        # Handle common prefixes
        prefixes = ['tbl', 'acc', 'mem', 'dep']
        for prefix in prefixes:
            if normalized.startswith(prefix):
                normalized = normalized[len(prefix):]
        
        return normalized

    def _find_table_variations(self, table_name: str) -> List[str]:
        """Generate possible variations of a table name."""
        variations = set()
        
        # Original name
        variations.add(table_name.lower())
        
        # Remove spaces
        variations.add(table_name.replace(' ', '').lower())
        
        # Add/remove underscores
        variations.add(table_name.replace(' ', '_').lower())
        variations.add(table_name.replace('_', ' ').lower())
        
        # Handle common prefixes
        if not table_name.lower().startswith('tbl'):
            variations.add('tbl' + table_name.replace(' ', '').lower())
        if not table_name.lower().startswith('acc'):
            variations.add('acc' + table_name.replace(' ', '').lower())
        
        # Handle camelCase
        camel_case = ''.join(word.capitalize() for word in table_name.split())
        variations.add(camel_case)
        
        return list(variations)

    async def _select_relevant_schemas(self, user_query: str, top_k: int = 5) -> List[Dict]:
        """Select relevant schemas based on query intensity and context."""
        try:
            if not user_query or not user_query.strip():
                logger.warning("Empty query provided.")
                return []

            cache_key = f"{self.db_name}:{user_query}:{top_k}"
            cached_schemas = self.cache_manager.get_cached_schema(cache_key)
            if cached_schemas:
                logger.info("Returning schemas from cache")
                return cached_schemas

            # Step 1: Get explicitly mentioned tables
            explicit_tables = self._extract_table_names(user_query)
            logger.info(f"📊 Explicitly mentioned tables: {explicit_tables}")

            # Step 2: Get all available tables and their metadata
            schema_docs = self.vector_store.collections["schema"].get()
            all_schemas = {}
            
            for doc in schema_docs["documents"]:
                try:
                    schema = json.loads(doc)
                    if isinstance(schema, dict) and "table_name" in schema:
                        all_schemas[schema["table_name"]] = schema
                except json.JSONDecodeError:
                    continue

            # Step 3: First, add explicitly mentioned tables
            final_schemas = []
            for table in explicit_tables:
                if table in all_schemas:
                    final_schemas.append(all_schemas[table])
                    logger.info(f"✅ Added explicitly mentioned table: {table}")

            # Step 4: Find and add related tables through foreign key relationships
            related_tables = set()
            for schema in final_schemas.copy():
                # Check foreign key relationships in the current table
                if "relationships" in schema:
                    for fk_col, ref in schema["relationships"].items():
                        ref_table = ref.split('(')[0]
                        if ref_table not in [s["table_name"] for s in final_schemas]:
                            related_tables.add(ref_table)
                            logger.info(f"✅ Found related table through FK: {ref_table}")

                # Check tables that reference this table
                for other_table, other_schema in all_schemas.items():
                    if other_table not in [s["table_name"] for s in final_schemas]:
                        for fk_col, ref in other_schema.get("relationships", {}).items():
                            if schema["table_name"] in ref:
                                related_tables.add(other_table)
                                logger.info(f"✅ Found table referencing current table: {other_table}")

            # Add related tables to final schemas
            for table in related_tables:
                if table in all_schemas:
                    final_schemas.append(all_schemas[table])
                    logger.info(f"✅ Added related table: {table}")

            # Step 5: If we still need more tables, use semantic search
            if len(final_schemas) < top_k and not explicit_tables:
                query_embedding = await self._get_embedding_async(user_query)
                results = self.vector_store.collections["schema"].query(
                    query_embeddings=[query_embedding],
                    n_results=top_k - len(final_schemas),
                    include=["documents"]
                )
                
                for doc in results["documents"][0] if results["documents"] else []:
                    try:
                        schema = json.loads(doc)
                        if (isinstance(schema, dict) and 
                            "table_name" in schema and 
                            schema["table_name"] not in [s["table_name"] for s in final_schemas]):
                            final_schemas.append(schema)
                            logger.info(f"✅ Added semantically relevant table: {schema['table_name']}")
                    except json.JSONDecodeError:
                        continue

            # Log final selection
            selected_tables = [schema["table_name"] for schema in final_schemas]
            logger.info(f"📊 Final selected tables: {selected_tables}")
            logger.info(f"📊 Found {len(final_schemas)} relevant schemas")

            self.cache_manager.cache_schema(cache_key, final_schemas)
            return final_schemas

        except Exception as e:
            logger.error(f"❌ Error selecting relevant schemas: {str(e)}")
            raise

    def _create_intent_aware_description(self, schema: Dict, intent: Dict) -> str:
        """Create a description of the table that's aware of the query intent."""
        desc = f"Table {schema['table_name']}"
        
        # Add columns based on intent
        if "columns" in schema:
            relevant_columns = []
            for col in schema["columns"]:
                col_name = col["column_name"].lower()
                
                # Check if column matches intent
                if intent["temporal_aspect"] and any(time_word in col_name for time_word in ["date", "time", "created", "updated"]):
                    relevant_columns.append(col["column_name"])
                elif intent["aggregation_level"] and any(agg_word in col_name for agg_word in ["amount", "total", "sum", "count"]):
                    relevant_columns.append(col["column_name"])
                elif col_name in intent["columns"]:
                    relevant_columns.append(col["column_name"])
                elif col.get("is_primary_key") or col.get("is_foreign_key"):
                    relevant_columns.append(col["column_name"])
            
            if relevant_columns:
                desc += " with columns: " + ", ".join(relevant_columns)
        
        # Add relationships based on intent
        if "relationships" in schema and intent["join_complexity"] != "none":
            desc += " related to: " + ", ".join(ref.split('(')[0] for ref in schema["relationships"].values())
        
        return desc

    def _cosine_similarity(self, v1: List[float], v2: List[float]) -> float:
        """Calculate cosine similarity between two vectors."""
        dot_product = sum(a * b for a, b in zip(v1, v2))
        norm_v1 = math.sqrt(sum(a * a for a in v1))
        norm_v2 = math.sqrt(sum(b * b for b in v2))
        return dot_product / (norm_v1 * norm_v2) if norm_v1 * norm_v2 != 0 else 0

    def _find_related_tables(self, tables: List[str]) -> List[str]:
        """Find tables related through foreign key relationships."""
        if not tables:
            return []
            
        related_tables = set()
        schema_docs = self.vector_store.collections["schema"].get()
        
        if not schema_docs["documents"]:
            logger.warning("No schema documents found")
            return []
            
        # Debug: Log all available tables and their relationships
        logger.info("📊 Available tables and their relationships:")
        for doc in schema_docs["documents"]:
            try:
                schema = json.loads(doc)
                if "table_name" in schema:
                    logger.info(f"Table: {schema['table_name']}")
                    if "relationships" in schema:
                        for fk_col, ref in schema["relationships"].items():
                            logger.info(f"  - FK: {fk_col} -> {ref}")
                    else:
                        logger.info("  - No relationships found")
            except json.JSONDecodeError:
                logger.error(f"Failed to parse schema document: {doc}")
                continue

        # Create a map of all tables and their relationships
        table_relationships = {}
        for doc in schema_docs["documents"]:
            try:
                schema = json.loads(doc)
                if "table_name" in schema:
                    table_name = schema["table_name"]
                    table_relationships[table_name] = []
                    
                    # Process direct relationships
                    if "relationships" in schema:
                        for fk_col, ref in schema["relationships"].items():
                            try:
                                ref_table = ref.split('(')[0]
                                ref_col = ref.split('(')[1].rstrip(')')
                                table_relationships[table_name].append({
                                    "type": "direct",
                                    "target_table": ref_table,
                                    "source_column": fk_col,
                                    "target_column": ref_col
                                })
                                logger.info(f"Found direct relationship: {table_name}.{fk_col} -> {ref_table}.{ref_col}")
                            except Exception as e:
                                logger.error(f"Error processing relationship {fk_col} -> {ref}: {str(e)}")
                    else:
                        logger.info(f"No relationships found for table {table_name}")
            except json.JSONDecodeError:
                logger.error(f"Failed to parse schema document: {doc}")
                continue

        # Debug: Log the relationship map
        logger.info("📊 Relationship map:")
        for table, rels in table_relationships.items():
            logger.info(f"Table {table} has {len(rels)} relationships:")
            for rel in rels:
                logger.info(f"  - {rel['type']}: {rel['source_column']} -> {rel['target_table']}.{rel['target_column']}")

        # Find all related tables
        for table in tables:
            if table in table_relationships:
                logger.info(f"Processing relationships for table: {table}")
                for rel in table_relationships[table]:
                    related_tables.add(rel["target_table"])
                    logger.info(f"Added related table: {rel['target_table']} (via {rel['type']} relationship)")
                    
                    # Also check if the related table has its own relationships
                    if rel["target_table"] in table_relationships:
                        for sub_rel in table_relationships[rel["target_table"]]:
                            if sub_rel["target_table"] not in tables:
                                related_tables.add(sub_rel["target_table"])
                                logger.info(f"Added indirectly related table: {sub_rel['target_table']}")

        # Log the final set of related tables
        logger.info(f"📊 Final set of related tables: {list(related_tables)}")
        
        return list(related_tables)

    def _extract_table_names(self, query: str) -> List[str]:
        if not query or not query.strip():
            logger.warning("Empty query provided.")
            return []

        query_lower = query.lower()
        try:
            table_docs = self.vector_store.collections["schema"].get()
            if not table_docs or not table_docs.get("documents"):
                logger.warning("No tables found in schema collection.")
                return []
            
            table_names = []
            for doc in table_docs["documents"]:
                try:
                    schema = json.loads(doc)
                    if isinstance(schema, dict) and "table_name" in schema:
                        table_names.append(schema["table_name"])
                except json.JSONDecodeError:
                    continue
            
            if not table_names:
                logger.warning("No valid table names found in schema collection.")
                return []

            mentioned_tables = []
            query_lower = re.sub(r'\s+', ' ', query_lower)
            sql_pattern = r'\b(?:from|join)\s+([A-Za-z_][A-Za-z0-9_]*)\b'
            sql_matches = re.findall(sql_pattern, query_lower, re.IGNORECASE)
            
            for match in sql_matches:
                if match in table_names:
                    mentioned_tables.append(match)
                    
            for table_name in table_names:
                if re.search(rf'\b{re.escape(table_name.lower())}\b', query_lower):
                    mentioned_tables.append(table_name)
                    continue
                
                synonyms = self._generate_table_synonyms(table_name)
                for synonym in synonyms:
                    if synonym and re.search(rf'\b{re.escape(synonym.lower())}\b', query_lower):
                        mentioned_tables.append(table_name)
                        break

            return list(set(mentioned_tables))
            
        except Exception as e:
            logger.error(f"❌ Error extracting table names: {str(e)}")
            return []

    def _select_relevant_columns(self, schema: Dict, query_intent: Dict) -> List[str]:
        relevant_columns = []
        relevant_columns.extend(schema.get("primary_keys", []))
        
        for col in schema.get("columns", []):
            col_name = col["column_name"].lower()
            if any(term in col_name for term in query_intent["key_terms"]):
                relevant_columns.append(col["column_name"])
                continue
                
            if query_intent["time_range"] and any(
                dtype in col.get("data_type", "").lower() 
                for dtype in ["date", "time", "timestamp"]
            ):
                relevant_columns.append(col["column_name"])
                continue
                
            if query_intent["action_type"] in ["sum", "average", "count"] and any(
                dtype in col.get("data_type", "").lower() 
                for dtype in ["int", "float", "decimal", "numeric"]
            ):
                relevant_columns.append(col["column_name"])
                continue
                
            if col.get("is_foreign_key"):
                relevant_columns.append(col["column_name"])
                
        return list(set(relevant_columns))

    def _score_table_relevance(self, table_doc: Dict, query_intent: Dict) -> float:
        score = 0.0
        table_name = table_doc.get("table_name", "").lower()
        table_desc = table_doc.get("description", "").lower()
        
        for term in query_intent["key_terms"]:
            term = term.lower()
            if term in table_name:
                score += 3.0
            elif term in table_name.replace("_", " "):
                score += 2.0
            if term in table_desc:
                score += 1.0
        
        for entity in query_intent["entities"]:
            entity = entity.lower()
            if entity in table_name or entity in table_desc:
                score += 2.5
        
        if "columns" in table_doc:
            for col in table_doc["columns"]:
                col_name = col["column_name"].lower()
                for term in query_intent["key_terms"]:
                    if term in col_name:
                        score += 1.5
                
                if query_intent["time_range"] and any(
                    dtype in col.get("data_type", "").lower() 
                    for dtype in ["date", "time", "timestamp"]
                ):
                    score += 2.0
                
                if query_intent["action_type"] in ["sum", "average"] and any(
                    dtype in col.get("data_type", "").lower() 
                    for dtype in ["int", "float", "decimal", "numeric"]
                ):
                    score += 1.5
        
        if table_doc.get("primary_keys") and query_intent["action_type"] == "select":
            score += 1.0
        
        return score

    def _schema_to_natural_language(self, schema: Dict) -> str:
        """Convert a schema JSON to a concise natural-language description with complete relationship information."""
        table_name = schema["table_name"]
        description = [f"Table: {table_name}"]
        
        # Add table description if available
        if "table_description" in schema:
            description.append(f"Description: {schema['table_description']}")
        
        # Add columns with complete information
        description.append("Columns:")
        for col in schema["columns"]:
            col_desc = [f"  - {col['column_name']} ({col['data_type']})"]
            attributes = []
            
            if col.get("is_primary_key"):
                attributes.append("Primary Key")
            if col.get("is_foreign_key"):
                attributes.append(f"Foreign Key to {col.get('foreign_key_reference', 'Unknown')}")
            if col.get("is_nullable"):
                attributes.append("Nullable")
            if col.get("default_value"):
                attributes.append(f"Default: {col['default_value']}")
                
            if attributes:
                col_desc.append(f" ({', '.join(attributes)})")
            description.append("".join(col_desc))
        
        # Add all relationships
        if schema.get("relationships"):
            description.append("\nRelationships:")
            for fk_col, ref in schema["relationships"].items():

                idx_desc = f"  - {idx['index_name']} on {idx['column']}"
                if idx.get("is_unique"):
                    idx_desc += " (UNIQUE)"
                description.append(idx_desc)
        
        return "\n".join(description)

    async def generate_sql(self, prompt: str, allow_llm_to_see_data: bool = False) -> str:
        """Generate SQL query from prompt."""
        try:
            cache_key = hashlib.md5(prompt.encode()).hexdigest()
            cached_result = self.cache_manager.get_cached_query(cache_key)
            if cached_result:
                logger.info("📎 Using cached SQL query")
                return cached_result

            # Run the synchronous generate_sql method in a thread pool
            loop = asyncio.get_event_loop()
            response = await loop.run_in_executor(
                self.executor,
                lambda: self.chat_model.generate_sql(prompt)
            )
            
            logger.info(f"📜 Raw LLM response:\n{response}")

            sql_query = response.strip()
            if sql_query.startswith("```sql"):
                sql_query = sql_query[6:]
            if sql_query.endswith("```"):
                sql_query = sql_query[:-3]
            
            sql_query = sql_query.strip()
            self.cache_manager.cache_query(cache_key, sql_query)
            return sql_query
            
        except Exception as e:
            logger.error(f"❌ Error in generate_sql: {str(e)}")
            raise

    async def generate_sql_query(self, user_query: str) -> str:
        """Generate an SQL query with improved accuracy and relevance for any table or column."""
        try:
            if not user_query or not user_query.strip():
                logger.error("Empty query provided")
                return "Error: Empty query provided."

            # Standardize the input query
            standardized_query = self._standardize_query(user_query)
            logger.info(f"📝 Standardized query: {standardized_query}")

            cache_key = hashlib.md5(standardized_query.encode()).hexdigest()
            cached_result = self.cache_manager.get_cached_query(cache_key)
            if cached_result:

                return cached_result

            try:

            except Exception as e:

                return f"Error: Failed to analyze query intent: {str(e)}"

            try:

            except Exception as e:
                logger.error(f"❌ Error selecting relevant schemas: {str(e)}")
                return f"Error: Failed to select relevant schemas: {str(e)}"

            # Build schema context with validation
            schema_descriptions = []
            for schema in schemas:
                try:

                    desc = [f"Table: {schema['table_name']}"]
                    desc.append(f"Description: {schema.get('table_description', 'No description available')}")

                    relevant_columns = self._select_relevant_columns(schema, query_intent)
                    if not relevant_columns and "details" in standardized_query.lower():
                        relevant_columns = [col["column_name"] for col in schema.get("columns", [])]
                    
                    if schema.get("columns"):
                        desc.append("Columns:")
                        for col in schema["columns"]:
                            if col["column_name"] in relevant_columns or col["is_primary_key"] or col["is_foreign_key"]:
                                col_desc = f"  - {col['column_name']} ({col['data_type']})"
                                if col["is_primary_key"]:
                                    col_desc += ", Primary Key"
                                if col["is_foreign_key"]:
                                    col_desc += f", Foreign Key to {col.get('foreign_key_reference', 'Unknown')}"
                                if col["is_nullable"]:
                                    col_desc += ", Nullable"
                                desc.append(col_desc)
                    
                    # Add relevant foreign key relationships with validation
                    if schema.get("relationships"):
                        desc.append("Foreign Key Relationships:")
                        for fk_col, ref in schema["relationships"].items():
                            if fk_col in relevant_columns or any(col in ref for col in relevant_columns):
                                desc.append(f"  - {fk_col} references {ref}")
                    
                    schema_descriptions.append("\n".join(desc))
                except Exception as e:
                    logger.error(f"❌ Error processing schema {schema.get('table_name', 'unknown')}: {str(e)}")
                    continue
            
            if not schema_descriptions:
                logger.error("No valid schema descriptions generated")
                return "Error: Failed to generate valid schema descriptions."
            
            schema_context = "\n\n".join(schema_descriptions)
            logger.info(f"📊 Generated schema context with {len(schema_descriptions)} tables")

            # Build deterministic prompt
            system_prompt = f"""
            You are an SQL expert for Microsoft SQL Server. Generate a valid, performant SQL query based on the provided schema and user question.

            1. You MUST use ONLY the exact tables, columns, and relationships shown in the schema below.

            {schema_context}

            2. Use explicit JOINs based on the foreign key relationships shown - no new joins allowed.
            3. Select only columns explicitly mentioned in the question or shown in the schema.
            4. Apply WHERE clauses for conditions specified in the question.

            VALIDATION:
            Your query will be strictly validated against the schema shown above. Any deviation will result in an error.
            - All table names must match exactly
            - All column names must match exactly
            - All joins must use existing foreign key relationships
            - No additional schema elements are allowed
            - Similar queries must follow the same pattern
            """

            user_prompt = f"""
            Database Schema (USE ONLY THESE EXACT ELEMENTS - NO EXCEPTIONS):
            {schema_context}

            User Question: {standardized_query}

            Requirements:
            - Use ONLY the tables, columns, and relationships shown in the schema above.
            - Include ONLY joins that are explicitly defined in the foreign key relationships.
            - Apply WHERE clauses for explicit conditions.
            - Use aggregations only if specified in the question.
            - Return only the SQL query.
            - If the question requires anything not in the schema, return an error message.
            - Follow a consistent pattern for similar queries.
            """

            full_prompt = f"{system_prompt}\n\n{user_prompt}"
            logger.info(f"📝 Sending prompt to LLM:\n{full_prompt}")

            # Generate SQL with retry logic
            max_retries = 3
            retry_count = 0
            last_error = None
            
            while retry_count < max_retries:
                try:
                    sql_query = await self.generate_sql(full_prompt, allow_llm_to_see_data=True)
                    if sql_query and sql_query.strip():
                        logger.info(f"✅ Successfully generated SQL query")

                    await asyncio.sleep(1)  # Add delay between retries
            
            if retry_count == max_retries:
                logger.error(f"❌ Failed to generate SQL query after {max_retries} retries: {str(last_error)}")
                return f"Error: Failed to generate SQL query after {max_retries} retries: {str(last_error)}"

            # Standardize the generated SQL
            standardized_sql = self._standardize_sql(sql_query)
            logger.info(f"🔍 Final standardized SQL: {standardized_sql}")
            
            # Cache the result
            self.cache_manager.cache_query(cache_key, standardized_sql)
            
            return standardized_sql
            
        except Exception as e:
            logger.error(f"❌ Error generating SQL query: {str(e)}")
            return f"Error: Failed to generate SQL query: {str(e)}"

    def _standardize_query(self, query: str) -> str:
        """Standardize the input query for consistent processing."""
        # Convert to lowercase
        query = query.lower()
        
        # Remove extra whitespace
        query = ' '.join(query.split())
        
        # Standardize common terms
        replacements = {
            'show me': 'show',
            'give me': 'show',
            'display': 'show',
            'list': 'show',
            'find': 'show',
            'get': 'show',
            'fetch': 'show',
            'retrieve': 'show',
            'where': 'where',
            'having': 'having',
            'group by': 'group by',
            'order by': 'order by',
            'limit': 'limit',
            'top': 'top',
            'count': 'count',
            'sum': 'sum',
            'avg': 'avg',
            'min': 'min',
            'max': 'max'
        }
        
        for old, new in replacements.items():
            query = query.replace(old, new)
        
        return query.strip()

    def _standardize_sql(self, sql: str) -> str:
        """Standardize the generated SQL for consistent output."""
        # Remove comments
        sql = re.sub(r'--.*$', '', sql, flags=re.MULTILINE)
        sql = re.sub(r'/\*.*?\*/', '', sql, flags=re.DOTALL)
        
        # Standardize whitespace
        sql = ' '.join(sql.split())
        
        # Standardize common SQL patterns
        sql = re.sub(r'\bSELECT\b', 'SELECT', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\bFROM\b', 'FROM', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\bWHERE\b', 'WHERE', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\bJOIN\b', 'JOIN', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\bON\b', 'ON', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\bGROUP BY\b', 'GROUP BY', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\bORDER BY\b', 'ORDER BY', sql, flags=re.IGNORECASE)
        
        # Standardize line breaks
        sql = re.sub(r'\s+', ' ', sql)
        sql = re.sub(r'SELECT\s+', 'SELECT\n  ', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\s+FROM\s+', '\nFROM\n  ', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\s+WHERE\s+', '\nWHERE\n  ', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\s+JOIN\s+', '\nJOIN\n  ', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\s+ON\s+', '\n  ON ', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\s+GROUP BY\s+', '\nGROUP BY\n  ', sql, flags=re.IGNORECASE)
        sql = re.sub(r'\s+ORDER BY\s+', '\nORDER BY\n  ', sql, flags=re.IGNORECASE)
        
        return sql.strip()

def format_schema_for_model(schema: Dict) -> str:
    """Format schema for optimal model consumption."""
    formatted_schema = []
    
    # Add table information with enhanced metadata
    for table in schema["tables"]:
        table_info = [
            f"Table: {table['name']}",
            f"Description: {table.get('description', 'No description available')}",
            f"Row Count: {table.get('row_count', 'Unknown')}",
            f"Last Updated: {table.get('last_updated', 'Unknown')}",
            "\nColumns:"
        ]
        
        # Format columns with enhanced information
        for col in table["columns"]:
            col_info = [
                f"- {col['name']} ({col['type']})",
                f"  Primary Key: {col.get('is_primary_key', False)}",
                f"  Nullable: {col.get('is_nullable', True)}",
                f"  Description: {col.get('description', 'No description')}",
                f"  Sample Values: {', '.join(str(v) for v in col.get('sample_values', []))}"
            ]
            table_info.extend(col_info)
        
        # Add relationships
        if table.get("relationships"):
            table_info.append("\nRelationships:")
            for rel in table["relationships"]:
                rel_info = [
                    f"- {rel['type']} relationship with {rel['target_table']}",
                    f"  Source Column: {rel['source_column']}",
                    f"  Target Column: {rel['target_column']}",
                    f"  Cardinality: {rel.get('cardinality', 'Unknown')}"
                ]
                table_info.extend(rel_info)
        
        formatted_schema.append("\n".join(table_info))
    
    return "\n\n".join(formatted_schema)
